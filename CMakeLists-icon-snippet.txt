# =========================================================================
# Icônes multiplateformes pour CollabLAN
# À insérer/adapter dans app/CMakeLists.txt (remplace l'add_executable actuel)
# =========================================================================

if(WIN32)
    # Le .rc référence icons/app.ico -> compilé dans l'exécutable
    set(APP_ICON_RESOURCE ${CMAKE_CURRENT_SOURCE_DIR}/../resources/win/app.rc)
elseif(APPLE)
    set(APP_ICON_RESOURCE ${CMAKE_CURRENT_SOURCE_DIR}/../resources/mac/app.icns)
    set_source_files_properties(${APP_ICON_RESOURCE} PROPERTIES
        MACOSX_PACKAGE_LOCATION "Resources"
    )
endif()

add_executable(CollabLAN
    main.cpp
    ../ui/qml.qrc
    ${APP_ICON_RESOURCE}
)

if(APPLE)
    set_target_properties(CollabLAN PROPERTIES
        MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_ICON_FILE app.icns
        MACOSX_BUNDLE_BUNDLE_NAME "CollabLAN"
        MACOSX_BUNDLE_GUI_IDENTIFIER "com.tonorg.collablan"
    )
endif()

target_link_libraries(CollabLAN PRIVATE
    Qt6::Quick
    Qt6::Qml
    Qt6::QuickControls2
    app_core
)

if(UNIX AND NOT APPLE)
    # Installation du .desktop + des icônes XDG (hicolor theme)
    install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/../resources/linux/collablan.desktop
            DESTINATION share/applications)
    foreach(size 16 24 32 48 64 128 256 512)
        install(FILES ${CMAKE_CURRENT_SOURCE_DIR}/../resources/linux/icons/${size}x${size}/collablan.png
                DESTINATION share/icons/hicolor/${size}x${size}/apps)
    endforeach()
endif()
